## Fix for Docusaurus

```tsx title="docusaurus.config.ts"
...
const config: Config = {
  ...
  plugins: [
    ...
    // highlight-start
    async function MMDDocusaurusFix(context, options) {
      return {
        name: "mmd-docusaurus-fix",
        configureWebpack: () => ({
          resolve: {
            fallback: { path: false, fs: false },
            extensions: [".jsx", ".js", ".tsx", ".ts"],
          },
        }),
      };
    },
    // highlight-end
    ...
  ],
  ...
  } satisfies Preset.ThemeConfig,
};
export default config;
```

<details>
<summary>Versions before 1.0.0: render only in the browser</summary>

Before 1.0.0, the player set `window.MMDLoadDetails` on import, so it failed during
server rendering and had to be wrapped to run only in the browser. In Docusaurus, use:


<Tabs groupId="language">
  <TabItem value="tsx" label="TypeScript" default>
  
```tsx title="MMDPlayerWrapper.tsx"
import BrowserOnly from "@docusaurus/BrowserOnly";
import type { MMD as MMDType } from "@arclockproject/mmd-player"; // Remove if on js
export function MMD(props: Parameters<typeof MMDType>) {
  return (
    <BrowserOnly fallback="Loading MMDPlayer...">
      {() => {
        const _MMD = require("@arclockproject/mmd-player").default;
        return <_MMD {...props} />;
      }}
    </BrowserOnly>
  );
}
```

  </TabItem>
