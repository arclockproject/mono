## Dependencies {#dependencies}

This library does not require any dependencies.

## Usage {#usage}

Play audio on button click.

```tsx
import { AudioManagerGlobal } from "@arclockproject/common/library/Audio";

function Example() {
  return (
    <div>
      <button
        onClick={() => {
          AudioManagerGlobal.play("/assets/sound.mp4");
        }}
      />
    </div>
  );
}
```

Play audio on element load and keep looping until element unmounts.

```tsx
import { AudioManagerGlobal } from "@arclockproject/common/library/Audio";
import { useEffect } from "react";

function Example() {
  useEffect(() => {
    AudioManagerGlobal.play("/assets/sound.mp4", {
      loop: true,
      singleInstance: true,
    });
    return () => {
      AudioManagerGlobal.stopInstance("/assets/sound.mp4", {
        gradual: true,
      });
    };
  }, []);
  return (
    <div>
      <button onClick={() => {}} />
    </div>
  );
}
```
